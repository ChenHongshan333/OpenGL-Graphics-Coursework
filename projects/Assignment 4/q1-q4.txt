#define MAXPTNO 1000
#define NLINESEGMENT 32
#define NOBJECTONCURVE 8

using namespace std;

// ------------------------------
// data structures
// ------------------------------
struct Point { int x, y; };
struct FPoint { float x, y; };

// store control points
int nPt = 0;
Point ptList[MAXPTNO];

// display options
bool displayControlPoints = true; // display control points by default
bool displayControlLines = true; // display control lines by default
bool displayTangentVectors = false; // don't display tangent vectors by default
bool displayObjects = false;        // don't display objects by default
bool C1Continuity = false; // don't enforce C1 continuity by default

// ------------------------------
// tools
// ------------------------------
static inline int curveCount() {
    if (nPt < 4) return 0;
    // grouping the curves：0..3, 3..6, 6..9 ...
    return 1 + (nPt - 4) / 3;
}


// Get the control points of the curve with index curveIdx; 
// if C1 is enabled, do a "replacement" of the second control point of that curve 
// (except for the first curve)
// meanwhile return the original P1 for gray display, and whether a replacement occurred
static inline void getCurveControlPoints(
    int curveIdx,
    Point& P0, Point& P1, Point& P2, Point& P3,
    bool useC1Adjusted,
    Point& origP1Out,
    bool& hasAdjusted)
{
    int i = curveIdx * 3;
    P0 = ptList[i];
    P1 = ptList[i + 1];
    P2 = ptList[i + 2];
    P3 = ptList[i + 3];

    origP1Out = P1;
    hasAdjusted = false;

    // C1：P1 = 2*P0 - Q
    // Q2 is the third point on the previous curve (i-1)
	// P0 is the shared endpoint (i)
    if (useC1Adjusted && curveIdx > 0) {
        Point Q2 = ptList[i - 1]; 
        Point Q3 = ptList[i];    
        (void)Q3; 

        Point newP1;
        newP1.x = 2 * P0.x - Q2.x;
        newP1.y = 2 * P0.y - Q2.y;
        P1 = newP1;
        hasAdjusted = true;
    }
}

// Calculate a point on a 3-degree Bézier curve at parameter t
static inline FPoint evalBezier(const Point& P0, const Point& P1, const Point& P2, const Point& P3, float t) {
    float u = 1.0f - t;
    float b0 = u * u * u;
    float b1 = 3 * u * u * t;
    float b2 = 3 * u * t * t;
    float b3 = t * t * t;
    FPoint r;
    r.x = b0 * P0.x + b1 * P1.x + b2 * P2.x + b3 * P3.x;
    r.y = b0 * P0.y + b1 * P1.y + b2 * P2.y + b3 * P3.y;
    return r;
}

// draw a right-pointing arrow at the origin along the positive X axis
void drawRightArrow()
{
    glColor3f(0, 1, 0);
    glBegin(GL_LINE_STRIP);
    glVertex2f(0, 0);
    glVertex2f(100, 0);
    glVertex2f(95, 5);
    glVertex2f(100, 0);
    glVertex2f(95, -5);
    glEnd();
}

// ------------------------------
// OpenGL rendering
// ------------------------------
void display(void)
{
    glClear(GL_COLOR_BUFFER_BIT | GL_DEPTH_BUFFER_BIT);
    glPushMatrix();

	// 1) control points (black)
    if (displayControlPoints)
    {
        glPointSize(5);
        glBegin(GL_POINTS);
        for (int i = 0; i < nPt; i++)
        {
            glColor3f(0, 0, 0);
            glVertex2d(ptList[i].x, ptList[i].y);
        }
        glEnd();
        glPointSize(1);
    }

	// 2) control lines (green)
	// but if C1 is enabled, 
    // use the "adjusted P1" for each curve to reflect the true control polygon
    if (displayControlLines && nPt >= 2)
    {
        glColor3f(0, 1, 0);
        int nCurves = curveCount();
        for (int c = 0; c < nCurves; ++c)
        {
            Point P0, P1, P2, P3, origP1; bool adjusted = false;
            getCurveControlPoints(c, P0, P1, P2, P3, C1Continuity, origP1, adjusted);

            glBegin(GL_LINE_STRIP);
            glVertex2d(P0.x, P0.y);
            glVertex2d(P1.x, P1.y);
            glVertex2d(P2.x, P2.y);
            glVertex2d(P3.x, P3.y);
            glEnd();

            // if replacement happens:
            // new P1: red
            // old P1: gray 
            if (adjusted)
            {
                glPointSize(6);
                glBegin(GL_POINTS);
                glColor3f(1, 0, 0);           // new P1 (effective)
                glVertex2d(P1.x, P1.y);
                glColor3f(0.7f, 0.7f, 0.7f); // old P1 (displayed)
                glVertex2d(origP1.x, origP1.y);
                glEnd();
                glPointSize(1);
            }
        }

		// if there are leftover points that don't form a full curve, draw them too
        if (nCurves == 0 && nPt >= 2) {
            glBegin(GL_LINE_STRIP);
            for (int i = 0; i < nPt; ++i) {
                glVertex2d(ptList[i].x, ptList[i].y);
            }
            glEnd();
        }
    }

    // 3) Bézier curve (blue)
    {
        int nCurves = curveCount();
        for (int c = 0; c < nCurves; ++c)
        {
            Point P0, P1, P2, P3, origP1; bool adjusted = false;
            getCurveControlPoints(c, P0, P1, P2, P3, C1Continuity, origP1, adjusted);

            glColor3f(0, 0, 1);
            glBegin(GL_LINE_STRIP);
            for (int k = 0; k <= NLINESEGMENT; ++k)
            {
                float t = (float)k / (float)NLINESEGMENT;
                FPoint q = evalBezier(P0, P1, P2, P3, t);
                glVertex2f(q.x, q.y);
            }
            glEnd();
        }
    }

    // 4) tangent vector
    // 5) objects

    glPopMatrix();
    glutSwapBuffers();
}

void reshape(int w, int h)
{
    glViewport(0, 0, (GLsizei)w, (GLsizei)h);
    glMatrixMode(GL_PROJECTION);
    glLoadIdentity();
    gluOrtho2D(0, w, h, 0);
    glMatrixMode(GL_MODELVIEW);
    glLoadIdentity();
}

void init(void)
{
    glClearColor(1.0, 1.0, 1.0, 1.0);
}

// ------------------------------
// File I/O
// ------------------------------
void readFile()
{
    std::ifstream file;
    file.open("savefile.txt");
    if (!file.is_open()) {
        cout << "Error: Cannot open savefile.txt for reading." << endl;
        return;
    }

    file >> nPt;

    if (nPt > MAXPTNO)
    {
        cout << "Error: File contains more than the maximum number of points." << endl;
        nPt = MAXPTNO;
    }

    for (int i = 0; i < nPt; i++)
    {
        file >> ptList[i].x;
        file >> ptList[i].y;
    }
    file.close();
}

void writeFile()
{
    std::ofstream file;
    file.open("savefile.txt");
    if (!file.is_open()) {
        cout << "Error: Cannot open savefile.txt for writing." << endl;
        return;
    }

    file << nPt << endl;
    for (int i = 0; i < nPt; i++)
    {
        file << ptList[i].x << " " << ptList[i].y << endl;
    }
    file.close();
}

// ------------------------------
// Interaction
// ------------------------------
void keyboard(unsigned char key, int x, int y)
{
    switch (key) {
    case 'r':
    case 'R':
        readFile();
        break;

    case 'w':
    case 'W':
        writeFile();
        break;

    case 'T':
    case 't':
        displayTangentVectors = !displayTangentVectors;
        break;

    case 'o':
    case 'O':
        displayObjects = !displayObjects;
        break;

    case 'p':
    case 'P':
        displayControlPoints = !displayControlPoints;
        break;

    case 'L':
    case 'l':
        displayControlLines = !displayControlLines;
        break;

    case 'C':
    case 'c':
        C1Continuity = !C1Continuity;
        break;

    case 'e':
    case 'E':
		// clear all points
        nPt = 0;
        break;

    case 'Q':
    case 'q':
        exit(0);
        break;

    default:
        break;
    }

    glutPostRedisplay();
}

void mouse(int button, int state, int x, int y)
{
    /* button: GLUT_LEFT_BUTTON, GLUT_MIDDLE_BUTTON, or GLUT_RIGHT_BUTTON */
    /* state: GLUT_UP or GLUT_DOWN */
    enum
    {
        MOUSE_LEFT_BUTTON = 0,
        MOUSE_MIDDLE_BUTTON = 1,
        MOUSE_RIGHT_BUTTON = 2,
        MOUSE_SCROLL_UP = 3,
        MOUSE_SCROLL_DOWN = 4
    };
    if ((button == MOUSE_LEFT_BUTTON) && (state == GLUT_UP))
    {
        if (nPt == MAXPTNO)
        {
            cout << "Error: Exceeded the maximum number of points." << endl;
            return;
        }
        ptList[nPt].x = x;
        ptList[nPt].y = y;
        nPt++;
    }
    glutPostRedisplay();
}

int main(int argc, char** argv)
{
    cout << "CS3241 Lab 4" << endl << endl;
    cout << "Left mouse click: Add a control point" << endl;
    cout << "Q: Quit" << endl;
    cout << "P: Toggle displaying control points" << endl;
    cout << "L: Toggle displaying control lines" << endl;
    cout << "E: Erase all points (Clear)" << endl;
    cout << "C: Toggle C1 continuity" << endl;
    cout << "T: Toggle displaying tangent vectors (placeholder)" << endl;
    cout << "O: Toggle displaying objects (placeholder)" << endl;
    cout << "R: Read in control points from \"savefile.txt\"" << endl;
    cout << "W: Write control points to \"savefile.txt\"" << endl;

    glutInit(&argc, argv);
    glutInitDisplayMode(GLUT_DOUBLE | GLUT_RGB | GLUT_DEPTH);
    glutInitWindowSize(600, 600);
    glutInitWindowPosition(50, 50);
    glutCreateWindow("CS3241 Assignment 4");

    init();
    glutDisplayFunc(display);
    glutReshapeFunc(reshape);
    glutMouseFunc(mouse);
    glutKeyboardFunc(keyboard);
    glutMainLoop();

    return 0;
}
